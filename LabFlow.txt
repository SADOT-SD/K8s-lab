Install istio
# 1. Set variables
$version = "1.23.2"
$zipUrl = "https://github.com/istio/istio/releases/download/$version/istio-$version-win.zip"
$zipFile = "istio-$version-win.zip"

# 2. Download the Istio Windows bundle
Invoke-WebRequest -Uri $zipUrl -OutFile $zipFile

# 3. Extract
Expand-Archive -Path $zipFile -DestinationPath .

# 4. Add istioctl to PATH (temporary for this session)
$istioPath = (Resolve-Path ".\istio-$version\bin").Path
$env:Path = "$istioPath;$env:Path"

# 5. Verify
istioctl version

#6. Connect to cluster
istioctl install -y --set profile=default
kubectl label namespace default istio-injection=enabled --overwrite=true
kubectl get pods -n istio-system



kubectl rollout restart deployment
